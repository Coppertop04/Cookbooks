years <- c("2401", "2402", "2403")
letters <- c("A", "B", "C", "D", "E", "F", "G", "H")
numbers <- sprintf("%02d", 01:12)

all_levels <- character()

for ( year in years) {
  for (letter in letters) {
    for (number in numbers) {
      level <- paste0(year, letter, number)
      all_levels <- c(all_levels, level)
    }
  }
}
